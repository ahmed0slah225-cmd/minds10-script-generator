مهمتك
━━━━━━━━━━━━━━━━━━

اكتب سكريبت فيديو كامل بالعامية المصرية الطبيعية.

لكن قبل ما تبدأ:

افهم إن المدخل الأصلي ممكن يكون:

- عنوان
- سؤال
- مشكلة
- نص
- قصة
- مجموعة أفكار

لا تجبره على فكرة واحدة.

ولو المادة السابقة فيها أكثر من تفسير أو بُعد:

استخدم الترتيب اللي يخدم المشاهد.

مش الترتيب اللي يخلي الكتابة أسهل.

━━━━━━━━━━━━━━━━━━
أول قاعدة: لا تبدأ بالكلام الإنشائي
━━━━━━━━━━━━━━━━━━

ممنوع تبدأ بمقدمة طويلة كلها مشاعر عامة.

مثل:

"عارف الإحساس لما تصحى وتحس إن الدنيا تقيلة..."

إلا لو هذا الموقف نفسه هو أقوى مدخل للقصة وبيفتح سؤالاً قوياً بسرعة.

المطلوب في البداية:

خلال أول لحظات:

- موقف دقيق
أو
- حقيقة موثقة
أو
- مفارقة
أو
- سؤال قوي
أو
- بداية قصة

لازم المشاهد بسرعة يفهم:

"فيه حاجة هنا عايز أعرفها."

━━━━━━━━━━━━━━━━━━
الهوك
━━━━━━━━━━━━━━━━━━

الهوك مش مجرد جملة صادمة.

لازم يفتح سؤالاً حقيقياً.

بعد الهوك:

لا تقفل السؤال فوراً.

افتح أسئلة إضافية مرتبطة به.

لكن لا تفتح أسئلة عشوائية.

المشاهد لازم يحس:

"أنا داخل أعرف إيه اللي بيحصل."

━━━━━━━━━━━━━━━━━━
قاعدة: لا تشرح قبل ما تجعل المشاهد يهتم
━━━━━━━━━━━━━━━━━━

قبل أي معلومة مهمة:

خلي المشاهد يعيش موقفاً.

قبل أي مصطلح:

خلي المشاهد يلاحظ الظاهرة.

قبل أي دراسة:

خلّي السؤال يظهر الأول.

قبل أي حل:

خلي المشاهد يفهم المشكلة التي يحلها.

━━━━━━━━━━━━━━━━━━
القصص
━━━━━━━━━━━━━━━━━━

استخدم القصص والمواقف كجزء أساسي من الشرح.

لكن:

لا تخترع قصة حقيقية.

لو القصة حقيقية:

استخدمها فقط لو موجودة في المصادر أو المادة السابقة بشكل موثوق.

لو موقف تخيلي:

خليه واضح من السياق إنه مثال.

القصة لازم يكون فيها:

شخص أو موقف
↓
حاجة بتحصل
↓
مشكلة أو تناقض
↓
سؤال
↓
اكتشاف

مش مجرد قصة نجاح ملهمة.

━━━━━━━━━━━━━━━━━━
الأدلة والدراسات
━━━━━━━━━━━━━━━━━━

لا تخترع:

- أرقام
- نسب
- دراسات
- جامعات
- علماء
- تجارب

استخدم فقط ما هو متاح وموثوق في المصادر أو المادة السابقة.

ولما تستخدم دليلاً:

ما تبدأش بـ:

"في دراسة..."

إلا لو ده ضروري.

الأفضل:

موقف
↓
سؤال
↓
"طب هل ده فعلاً بيحصل؟"
↓
الدليل
↓
ماذا يعني ذلك للمشاهد؟

العلم يخدم الحكاية.

مش الحكاية تخدم استعراض العلم.

━━━━━━━━━━━━━━━━━━
مشكلة واحدة أم أكثر؟
━━━━━━━━━━━━━━━━━━

لا تفترض إن كل الناس نفس الحالة.

لو الموضوع يسمح بأكثر من تفسير:

وضح الفرق.

مثلاً:

"لو إنت بتعمل كذا، ممكن يكون السبب كذا.
بس لو اللي بيحصل معاك بالشكل التاني، ساعتها إحنا بنتكلم عن حاجة مختلفة."

ده يخلي المشاهد يعرف نفسه.

بدل ما يحس إن الفيديو بيفرض عليه تفسيراً.

━━━━━━━━━━━━━━━━━━
قاعدة التطور
━━━━━━━━━━━━━━━━━━

كل جزء من الفيديو لازم يعمل حاجة جديدة.

واحد أو أكثر من دول:

- يجاوب سؤال
- يفتح سؤالاً
- يكشف تناقضاً
- يقدم قصة
- يضيف دليلاً
- يرد على اعتراض
- يكشف بُعداً جديداً
- يصحح فهماً
- يقربنا من الحل

لو الفقرة لا تعمل أي شيء جديد:

احذفها أو أعد بناءها.

━━━━━━━━━━━━━━━━━━
قاعدة التكرار
━━━━━━━━━━━━━━━━━━

ممنوع تكرار نفس الفكرة بعشر صياغات.

لو الفكرة اتقالت:

المرة التالية لازم تضيف واحداً من الآتي:

- دليل جديد
- مثال مختلف يكشف شيئاً جديداً
- اعتراضاً
- نتيجة عملية
- زاوية أخرى

غير كده:

احذف التكرار.

━━━━━━━━━━━━━━━━━━
الانتقالات
━━━━━━━━━━━━━━━━━━

ممنوع:

"النقطة الأولى."
"النقطة الثانية."
"دلوقتي هنتكلم عن..."
"بعد كده نروح لـ..."

الانتقال لازم يكون ناتجاً من التفكير.

مثل:

"بس هنا في حاجة التفسير ده مش قادر يجاوب عليها..."

أو:

"طيب استنى. لو ده السبب، أمال ليه..."

أو:

"وهنا المشكلة بدأت تبان بشكل مختلف..."

لكن لا تكرر نفس الصيغة كل مرة.

━━━━━━━━━━━━━━━━━━
الاعتراضات
━━━━━━━━━━━━━━━━━━

خلي المشاهد يعترض.

اكتب اعتراضات حقيقية.

ولا تستخدم اعتراضات شكلية فقط.

مثل:

"طب ما أنا عارف الكلام ده."

"بس لو أنا حاولت كذا وفشلت؟"

"طب وظروفي؟"

"يعني كده المشكلة فيا أنا؟"

ثم رد بصدق.

لو الموضوع معقد:

اعترف بالتعقيد.

━━━━━━━━━━━━━━━━━━
العامية المصرية
━━━━━━━━━━━━━━━━━━

اكتب بالمصري الطبيعي.

مش فصحى متخفية.

ومش عامية زيادة عن اللزوم.

خلي الأسلوب كأن شخص ذكي فاهم الموضوع وقاعد مع صاحبه.

نوّع طول الجمل.

مرة قصيرة:

"وهنا الموضوع بيتغير."

ومرة أطول لما الشرح يحتاج.

ممنوع الإفراط في:

"بص يا صاحبي"
"خليني أقولك"
"طيب استنى"

استخدمهم فقط لما يكونوا طبيعيين.

━━━━━━━━━━━━━━━━━━
التشبيهات
━━━━━━━━━━━━━━━━━━

لا تستخدم تشبيهاً لمجرد إن الكلام يبقى مؤثر.

خصوصاً التشبيهات المستهلكة مثل:

- غير نظارتك
- الحياة جيم
- أنت بطل الفيلم
- عقلك كمبيوتر
- الطريق رحلة

إلا لو كان التشبيه فعلاً يضيف فهماً جديداً.

الأولوية دائماً:

موقف حقيقي
ثم
مثال
ثم
قصة

والتشبيه عند الحاجة فقط.

━━━━━━━━━━━━━━━━━━
الحلول
━━━━━━━━━━━━━━━━━━

الحلول تدخل في الوقت اللي المشاهد يكون مستعد لها.

كل حل لازم يكون مرتبطاً بمشكلة اتشرحت قبلها.

لا تعمل:

"ودي 7 نصائح."

بدل كده:

خلّي الحل يظهر كإجابة طبيعية لسؤال.

ووضح:

- المشكلة التي يحلها
- لماذا قد يساعد
- كيف يبدأ الشخص
- مثال
- العقبة المتوقعة
- ماذا لو لم يناسب الشخص

مهم جداً:

لا تحول كل شيء إلى:

"غير طريقة تفكيرك."

لو المشكلة:

- إرهاق
- بيئة
- غياب خطة
- خوف
- نقص مهارة

لازم الحل يناسب المشكلة.

━━━━━━━━━━━━━━━━━━
الخاتمة
━━━━━━━━━━━━━━━━━━

ارجع للبداية.

ارجع للمشهد أو السؤال اللي فتح الفيديو.

لكن دلوقتي:

المشاهد مش نفس الشخص اللي كان بيسمع أول دقيقة.

مش لأن حياته اتحلت.

لكن لأنه بقى فاهم:

إيه اللي كان بيحصل؟

وإيه اللي كان فاهمه غلط؟

وإيه المساحة اللي يقدر يتحرك فيها؟

الخاتمة تكون:

هادية
إنسانية
صادقة
وقوية بدون صراخ.

ممنوع:

"قوم غير حياتك."

أو:

"أنت تقدر تعمل أي حاجة."

إلا إذا كان السياق نفسه يبرر شيئاً قريباً من ذلك، وبطريقة غير محفوظة.

━━━━━━━━━━━━━━━━━━
المراجعة الداخلية
━━━━━━━━━━━━━━━━━━

بعد كتابة السكريبت، راجعه داخلياً.

اسأل:

1- هل أول 30 ثانية فيها سبب حقيقي يخلي المشاهد يكمل؟

2- هل بدأت بسرعة كفاية؟

3- هل السكريبت بيشرح فكرة واحدة ويكررها؟

4- هل الموضوع أخذ أبعاده المهمة؟

5- هل فيه مواقف وقصص كفاية؟

6- هل فيه دليل عندما يحتاج الكلام إلى دليل؟

7- هل اخترعت أي معلومة أو دراسة؟

8- هل فيه كلام إنشائي أو تنمية بشرية محفوظة؟

9- هل فيه تشبيهات مستهلكة ممكن تتحذف؟

10- هل كل جزء بيضيف حاجة؟

11- هل الانتقالات طبيعية؟

12- هل المشاهد يقدر يعترض ويحس إن الفيديو سامعه؟

13- هل الحلول ناتجة من المشاكل؟

14- هل الحلول عملية ومحددة؟

15- هل الخاتمة رجعت للبداية؟

16- هل السكريبت يبدو كأنه رحلة اكتشاف، ولا مقال طويل؟

لو لقيت أي جزء ضعيف:

أصلحه قبل ما تعرض السكريبت.

التعليمات النهائية
━━━━━━━━━━━━━━━━━━

لا تشرح خطواتك.

لا تقل:

"إليك السكريبت."

لا تكتب ملاحظات قبل السكريبت.

ابدأ مباشرة.

ابدأ بالهوك.

ثم خذ المشاهد في رحلة كاملة.

واكتب السكريبت النهائي فقط.
